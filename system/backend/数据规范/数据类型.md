# 数据类型

## patients

```json
{
  "patient_id": "p1",               // 唯一病人ID（全系统唯一标识一个人）
  "patient_code": "P001",           // 展示用编号（前端显示）
  "name": "Zhang San",              // 姓名
  "gender": "male",                 // 性别（male / female / other）
  "age": 68,                        // 年龄
  "date_of_birth": "1958-03-12",    // 出生日期（可选）
  "baseline_profile": {             // 基础健康背景（JSON结构）
    "comorbidities": ["hypertension", "diabetes"],
    "allergies": ["penicillin"]
  },
  "created_at": "timestamp",        // 创建时间
  "updated_at": "timestamp"         // 更新时间
}
```

------

## beds

```json
{
  "bed_id": "b1",                   // 唯一床位ID
  "bed_code": "ICU-01",             // 展示编号
  "room_code": "Room-A",            // 所在房间/区域
  "bed_type": "standard",           // 床位类型（standard / isolation 等）
  "status": "occupied",             // 当前状态（occupied / empty / cleaning / maintenance）
  "notes": null,                   // 备注信息
  "created_at": "timestamp",        // 创建时间
  "updated_at": "timestamp"         // 更新时间
}
```

------

## admissions

```json
{
  "admission_id": "ICU-ADM-0001",   // 一次ICU住院唯一ID
  "patient_id": "p1",               // 对应病人ID
  "bed_id": "b1",                   // 当前床位ID
  "admission_code": "ADM-001",      // 展示编号
  "admit_time": "timestamp",        // 入ICU时间
  "discharge_time": null,           // 出ICU时间（未出院为null）
  "status": "active",               // 状态（active / discharged / expired / transferred）
  "primary_diagnosis": "septic shock", // 主要诊断
  "admission_reason": "ED transfer due to hypotension", // 入ICU原因说明
  "severity_on_admission": "critical", // 入科严重程度（stable / unstable / critical）
  "attending_team": "ICU Team A",   // 负责团队
  "scenario_tag": "sepsis_progressive", // 模拟场景标签
  "created_at": "timestamp",        // 创建时间
  "updated_at": "timestamp"         // 更新时间
}
```

## events

```json
{
  "event_id": "evt1",               // 事件唯一ID
  "admission_id": "ICU-ADM-0001",   // 所属住院ID
  "patient_id": "p1",               // 病人ID
  "bed_id": "b1",                   // 床位ID
  "event_type": "vital_sign",       // 事件类型（vital_sign / lab / intervention / agent_output）
  "source": "monitor",              // 来源（monitor / lab / nurse / agent）
  "timestamp": "timestamp",         // 发生时间
  "priority": "high",               // 优先级（low / normal / high / critical）
  "payload": {}                     // 原始数据内容（JSON）
}
```

------

## vital_sign_events

```json
{
  "id": "vital1",                     // 生命体征记录ID
  "admission_id": "ICU-ADM-0001",     // 所属住院ID
  "timestamp": "timestamp",           // 时间

  "heart_rate": 120,                  // 心率（次/min）【APACHE】
  "mean_arterial_pressure": 60,       // 平均动脉压 MAP（mmHg）【APACHE】

  "systolic_bp": 90,                  // 收缩压（mmHg）
  "diastolic_bp": 50,                 // 舒张压（mmHg）

  "respiratory_rate": 24,             // 呼吸频率（次/min）【APACHE】

  "temperature": 38.2,                // 体温（°C）【APACHE】

  "spo2": 89,                         // 血氧饱和度（%）

  "fio2": 0.6,                        // 吸氧浓度 FiO2（0-1）【用于氧合评估】

  "pao2": 65,                         // 动脉氧分压（mmHg）【APACHE 5b】
  "aado2": null,                      // A-aDO2（氧梯度）【APACHE 5a】

  "ph": 7.28,                         // 动脉血pH【APACHE】

  "gcs": 12                           // Glasgow昏迷评分【APACHE】
}
```


## intervention_events

```json
{
  "id": "intv1",                  // 干预记录ID
  "admission_id": "ICU-ADM-0001", // 所属住院ID
  "timestamp": "timestamp",       // 时间
  "intervention_type": "fluid",   // 干预类型（fluid / vasopressor / ventilator_change）
  "description": "500ml saline",  // 干预描述
  "dosage": 500,                  // 剂量（可选）
  "unit": "ml"                    // 单位（可选）
}
```

## patient_state_current

```json
{
  "admission_id": "ICU-ADM-0001", // 主键（当前住院）
  "patient_id": "p1",             // 病人ID
  "bed_id": "b1",                 // 床位ID
  "updated_at": "timestamp",      // 更新时间

  "current_vitals": {},           // 当前生命体征（JSON汇总）
  "active_problems": [],          // 当前问题列表
  "active_risks": [],             // 当前风险列表
  "latest_interventions": [],     // 最近干预

  "care_phase": "critical"        // 当前阶段（stable / unstable / critical）
}
```

## risk_assessments

```json
{
  "id": "risk1",                // 风险评估ID
  "admission_id": "ICU-ADM-0001", // 所属住院ID
  "timestamp": "timestamp",     // 时间

  "risk_type": "shock",         // 风险类型
  "confidence": 0.92,           // 置信度（0-1）
  "severity": "critical",       // 严重程度（low / warning / critical）

  "evidence": [],               // 证据（JSON列表）
  "time_window": "last_2h",     // 时间窗口
  "recommended_action": "review fluid response" // 建议关注点
}
```

## alerts

```json
{
  "alert_id": "alert1",         // 告警ID
  "admission_id": "ICU-ADM-0001", // 所属住院ID
  "patient_id": "p1",           // 病人ID
  "bed_id": "b1",               // 床位ID

  "alert_type": "shock_risk",   // 告警类型
  "severity": "critical",       // 严重程度（info / warning / critical）

  "status": "open",             // 状态（open / acknowledged / closed）
  "source_agent": "risk_sentinel", // 来源agent

  "evidence": [],               // 证据
  "first_seen_at": "timestamp", // 首次出现时间
  "last_seen_at": "timestamp"   // 最近更新时间
}
```

## audit_logs

```json
{
  "id": "log1",                // 日志ID
  "timestamp": "timestamp",    // 时间

  "actor": "agent",            // 执行者类型（agent / system / user）
  "actor_id": "risk_sentinel", // 执行者ID

  "action_type": "run_agent",  // 行为类型（create_event / update_state / run_agent）
  "target_type": "risk",       // 作用对象类型
  "target_id": "risk1",        // 作用对象ID

  "input": {},                 // 输入数据
  "output": {}                 // 输出数据
}
```
